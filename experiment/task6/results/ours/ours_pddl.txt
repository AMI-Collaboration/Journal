(define (problem home-training-setup)
  (:domain household-multiagent)
  (:objects
    r1 - gripper
    r2 - mobile_heavy
    r3 - mobile_light
    r4 - mobile_light
    kitchen living_room bedroom bathroom - room
    mug coffeetable handtowel floorlamp - item
  )
  (:init
    (agent-at r1 kitchen)
    (agent-at r2 living_room)
    (agent-at r3 bedroom)
    (agent-at r4 bathroom)
    (agent-free r1) (agent-free r2) (agent-free r3) (agent-free r4)
    (can-access r2 kitchen) (can-access r2 living_room) (can-access r2 bathroom)
    (can-access r3 bedroom)
    (can-access r4 bathroom) (can-access r4 living_room)
    (object-at mug kitchen)
    (object-at coffeetable living_room)
    (object-at handtowel bathroom)
    (object-at floorlamp living_room)
  )
  (:goal
    (and
      (is-clean coffeetable)
      (object-at mug living_room)
      (object-at handtowel living_room)
      (is-on floorlamp)
    )
  )
)
